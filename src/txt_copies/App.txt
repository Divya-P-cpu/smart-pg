import { useEffect, useMemo, useState } from "react";
import PGCard from "./components/PGCard";
import { fetchPGs } from "./utils/pgApi";
import { AMENITIES, AREAS, CITIES } from "./utils/constants";

function App() {
  const [pgs, setPgs] = useState([]);

  const [loading, setLoading] = useState(false);
  const [error, setError] = useState("");

  const [savedPGs, setSavedPGs] = useState([]);
  const [comparedPGs, setComparedPGs] = useState([]);

  const [city, setCity] = useState("");
  const [area, setArea] = useState("");

  const [minBudget, setMinBudget] = useState("");
  const [maxBudget, setMaxBudget] = useState("");

  const [gender, setGender] = useState("");
  const [amenity, setAmenity] = useState("");
  const [sharing, setSharing] = useState("");

  const [showCitySuggestions, setShowCitySuggestions] = useState(false);
  const [showAreaSuggestions, setShowAreaSuggestions] = useState(false);

  const [hasSearched, setHasSearched] = useState(false);

  const citySuggestions = useMemo(() => {
    const value = city.trim().toLowerCase();

    if (!value) {
      return CITIES;
    }

    return CITIES.filter((item) =>
      item.toLowerCase().includes(value)
    );
  }, [city]);

  const areaSuggestions = useMemo(() => {
    const value = area.trim().toLowerCase();

    let filteredAreas = AREAS;

    if (city.trim()) {
      filteredAreas = filteredAreas.filter(
        (item) =>
          item.city.toLowerCase() === city.trim().toLowerCase()
      );
    }

    if (!value) {
      return filteredAreas.slice(0, 8);
    }

    return filteredAreas
      .filter((item) =>
        item.name.toLowerCase().includes(value)
      )
      .slice(0, 8);
  }, [city, area]);

  useEffect(() => {
    loadPGs();
  }, []);

  async function loadPGs(filters = {}) {
    try {
      setLoading(true);
      setError("");

      const data = await fetchPGs({
        city: filters.city !== undefined ? filters.city : city,
        area: filters.area !== undefined ? filters.area : area,
        min_budget:
          filters.min_budget !== undefined
            ? filters.min_budget
            : minBudget,
        max_rent:
          filters.max_rent !== undefined
            ? filters.max_rent
            : maxBudget,
        gender_policy:
          filters.gender_policy !== undefined
            ? filters.gender_policy
            : gender,
        amenity:
          filters.amenity !== undefined
            ? filters.amenity
            : amenity,
        sharing:
          filters.sharing !== undefined
            ? filters.sharing
            : sharing,
        page: 1,
        page_size: 100,
      });

      setPgs(Array.isArray(data.items) ? data.items : []);
    } catch (err) {
      console.error("Unable to load PGs:", err);
      setError(
        "Unable to load PGs. Please check whether the backend server is running."
      );
      setPgs([]);
    } finally {
      setLoading(false);
    }
  }

  function handleSearch() {
    setShowCitySuggestions(false);
    setShowAreaSuggestions(false);
    setHasSearched(true);

    loadPGs();
  }

  function handleReset() {
    setCity("");
    setArea("");
    setMinBudget("");
    setMaxBudget("");
    setGender("");
    setAmenity("");
    setSharing("");
    setHasSearched(false);

    loadPGs({
      city: "",
      area: "",
      min_budget: "",
      max_rent: "",
      gender_policy: "",
      amenity: "",
      sharing: "",
    });
  }

  function selectCity(selectedCity) {
    setCity(selectedCity);
    setArea("");
    setShowCitySuggestions(false);
    setShowAreaSuggestions(false);
  }

  function selectArea(selectedArea) {
    setArea(selectedArea);
    setShowAreaSuggestions(false);
  }

  function toggleSave(pgId) {
    setSavedPGs((current) =>
      current.includes(pgId)
        ? current.filter((id) => id !== pgId)
        : [...current, pgId]
    );
  }

  function toggleCompare(pgId) {
    setComparedPGs((current) =>
      current.includes(pgId)
        ? current.filter((id) => id !== pgId)
        : [...current, pgId]
    );
  }

  function viewDetails(pgId) {
    console.log("View PG details:", pgId);
  }

  return (
    <div className="app">
      <header
        style={{
          padding: "20px",
          borderBottom: "1px solid var(--border)",
          background: "#fff",
        }}
      >
        <h1
          style={{
            margin: 0,
            fontSize: "24px",
          }}
        >
          Smart PG
        </h1>

        <p
          style={{
            margin: "5px 0 0",
            color: "var(--muted)",
          }}
        >
          Find your perfect PG
        </p>
      </header>

      <main
        style={{
          maxWidth: "1200px",
          margin: "0 auto",
          padding: "24px 16px",
        }}
      >
        <section
          className="card"
          style={{
            padding: "20px",
            marginBottom: "24px",
          }}
        >
          <h2
            style={{
              margin: "0 0 16px",
              fontSize: "18px",
            }}
          >
            Find a PG
          </h2>

          <div
            style={{
              display: "grid",
              gridTemplateColumns:
                "repeat(auto-fit, minmax(210px, 1fr))",
              gap: "14px",
            }}
          >
            <div style={{ position: "relative" }}>
              <label
                style={{
                  display: "block",
                  marginBottom: "6px",
                  fontSize: "12px",
                  fontWeight: 600,
                }}
              >
                City
              </label>

              <input
                type="text"
                value={city}
                placeholder="Search city"
                onChange={(event) => {
                  setCity(event.target.value);
                  setArea("");
                  setShowCitySuggestions(true);
                }}
                onFocus={() => setShowCitySuggestions(true)}
                style={{
                  width: "100%",
                  boxSizing: "border-box",
                  padding: "10px 12px",
                  border: "1px solid var(--border)",
                  borderRadius: "8px",
                  outline: "none",
                }}
              />

              {showCitySuggestions &&
                citySuggestions.length > 0 && (
                  <div
                    style={{
                      position: "absolute",
                      top: "68px",
                      left: 0,
                      right: 0,
                      zIndex: 20,
                      background: "#fff",
                      border: "1px solid var(--border)",
                      borderRadius: "8px",
                      boxShadow:
                        "0 8px 20px rgba(0,0,0,0.12)",
                      overflow: "hidden",
                    }}
                  >
                    {citySuggestions.map((suggestion) => (
                      <button
                        key={suggestion}
                        type="button"
                        onClick={() => selectCity(suggestion)}
                        style={{
                          width: "100%",
                          padding: "10px 12px",
                          border: "none",
                          background: "#fff",
                          textAlign: "left",
                          cursor: "pointer",
                          fontSize: "13px",
                        }}
                      >
                        <i
                          className="fas fa-location-dot"
                          style={{
                            marginRight: "8px",
                            color: "var(--accent)",
                          }}
                        ></i>
                        {suggestion}
                      </button>
                    ))}
                  </div>
                )}
            </div>

            <div style={{ position: "relative" }}>
              <label
                style={{
                  display: "block",
                  marginBottom: "6px",
                  fontSize: "12px",
                  fontWeight: 600,
                }}
              >
                Area
              </label>

              <input
                type="text"
                value={area}
                placeholder={
                  city
                    ? "Search area"
                    : "Select city first"
                }
                disabled={!city}
                onChange={(event) => {
                  setArea(event.target.value);
                  setShowAreaSuggestions(true);
                }}
                onFocus={() => setShowAreaSuggestions(true)}
                style={{
                  width: "100%",
                  boxSizing: "border-box",
                  padding: "10px 12px",
                  border: "1px solid var(--border)",
                  borderRadius: "8px",
                  outline: "none",
                  background: city ? "#fff" : "#f5f5f5",
                }}
              />

              {showAreaSuggestions &&
                city &&
                areaSuggestions.length > 0 && (
                  <div
                    style={{
                      position: "absolute",
                      top: "68px",
                      left: 0,
                      right: 0,
                      zIndex: 20,
                      background: "#fff",
                      border: "1px solid var(--border)",
                      borderRadius: "8px",
                      boxShadow:
                        "0 8px 20px rgba(0,0,0,0.12)",
                      overflow: "hidden",
                      maxHeight: "280px",
                      overflowY: "auto",
                    }}
                  >
                    {areaSuggestions.map((suggestion) => (
                      <button
                        key={`${suggestion.city}-${suggestion.name}`}
                        type="button"
                        onClick={() =>
                          selectArea(suggestion.name)
                        }
                        style={{
                          width: "100%",
                          padding: "10px 12px",
                          border: "none",
                          background: "#fff",
                          textAlign: "left",
                          cursor: "pointer",
                          fontSize: "13px",
                        }}
                      >
                        <i
                          className="fas fa-location-dot"
                          style={{
                            marginRight: "8px",
                            color: "var(--accent)",
                          }}
                        ></i>

                        {suggestion.name}
                      </button>
                    ))}
                  </div>
                )}
            </div>

            <div>
              <label
                style={{
                  display: "block",
                  marginBottom: "6px",
                  fontSize: "12px",
                  fontWeight: 600,
                }}
              >
                Minimum Budget
              </label>

              <input
                type="number"
                min="0"
                value={minBudget}
                placeholder="₹ Minimum"
                onChange={(event) =>
                  setMinBudget(event.target.value)
                }
                style={{
                  width: "100%",
                  boxSizing: "border-box",
                  padding: "10px 12px",
                  border: "1px solid var(--border)",
                  borderRadius: "8px",
                  outline: "none",
                }}
              />
            </div>

            <div>
              <label
                style={{
                  display: "block",
                  marginBottom: "6px",
                  fontSize: "12px",
                  fontWeight: 600,
                }}
              >
                Maximum Budget
              </label>

              <input
                type="number"
                min="0"
                value={maxBudget}
                placeholder="₹ Maximum"
                onChange={(event) =>
                  setMaxBudget(event.target.value)
                }
                style={{
                  width: "100%",
                  boxSizing: "border-box",
                  padding: "10px 12px",
                  border: "1px solid var(--border)",
                  borderRadius: "8px",
                  outline: "none",
                }}
              />
            </div>

            <div>
              <label
                style={{
                  display: "block",
                  marginBottom: "6px",
                  fontSize: "12px",
                  fontWeight: 600,
                }}
              >
                Gender
              </label>

              <select
                value={gender}
                onChange={(event) =>
                  setGender(event.target.value)
                }
                style={{
                  width: "100%",
                  boxSizing: "border-box",
                  padding: "10px 12px",
                  border: "1px solid var(--border)",
                  borderRadius: "8px",
                  outline: "none",
                  background: "#fff",
                }}
              >
                <option value="">All Genders</option>
                <option value="Male">Male</option>
                <option value="Female">Female</option>
                <option value="Unisex">Unisex</option>
              </select>
            </div>

            <div>
              <label
                style={{
                  display: "block",
                  marginBottom: "6px",
                  fontSize: "12px",
                  fontWeight: 600,
                }}
              >
                Sharing
              </label>

              <select
                value={sharing}
                onChange={(event) =>
                  setSharing(event.target.value)
                }
                style={{
                  width: "100%",
                  boxSizing: "border-box",
                  padding: "10px 12px",
                  border: "1px solid var(--border)",
                  borderRadius: "8px",
                  outline: "none",
                  background: "#fff",
                }}
              >
                <option value="">Any Sharing</option>
                <option value="1">Single Sharing</option>
                <option value="2">2 Sharing</option>
                <option value="3">3 Sharing</option>
                <option value="4">4 Sharing</option>
                <option value="5">5 Sharing</option>
              </select>
            </div>

            <div>
              <label
                style={{
                  display: "block",
                  marginBottom: "6px",
                  fontSize: "12px",
                  fontWeight: 600,
                }}
              >
                Amenity
              </label>

              <select
                value={amenity}
                onChange={(event) =>
                  setAmenity(event.target.value)
                }
                style={{
                  width: "100%",
                  boxSizing: "border-box",
                  padding: "10px 12px",
                  border: "1px solid var(--border)",
                  borderRadius: "8px",
                  outline: "none",
                  background: "#fff",
                }}
              >
                <option value="">Any Amenity</option>

                {AMENITIES.map((item) => (
                  <option
                    key={item.value}
                    value={item.value}
                  >
                    {item.label}
                  </option>
                ))}
              </select>
            </div>
          </div>

          <div
            style={{
              display: "flex",
              justifyContent: "flex-end",
              gap: "10px",
              marginTop: "18px",
              flexWrap: "wrap",
            }}
          >
            <button
              type="button"
              onClick={handleReset}
              style={{
                padding: "10px 18px",
                border: "1px solid var(--border)",
                borderRadius: "8px",
                background: "#fff",
                cursor: "pointer",
                fontWeight: 600,
              }}
            >
              Reset
            </button>

            <button
              type="button"
              onClick={handleSearch}
              disabled={loading}
              style={{
                padding: "10px 20px",
                border: "none",
                borderRadius: "8px",
                background: "var(--accent)",
                color: "#fff",
                cursor: loading
                  ? "not-allowed"
                  : "pointer",
                fontWeight: 600,
                opacity: loading ? 0.7 : 1,
              }}
            >
              {loading ? "Searching..." : "Search PGs"}
            </button>
          </div>
        </section>

        <div
          style={{
            display: "flex",
            justifyContent: "space-between",
            alignItems: "center",
            marginBottom: "20px",
            gap: "12px",
            flexWrap: "wrap",
          }}
        >
          <div>
            <h2
              style={{
                margin: 0,
                fontSize: "20px",
              }}
            >
              {hasSearched ? "Search Results" : "Available PGs"}
            </h2>

            <p
              style={{
                margin: "5px 0 0",
                color: "var(--muted)",
                fontSize: "12px",
              }}
            >
              Total PGs: {pgs.length}
            </p>
          </div>

          <button
            type="button"
            onClick={() => loadPGs()}
            disabled={loading}
            style={{
              padding: "8px 14px",
              border: "1px solid var(--border)",
              borderRadius: "8px",
              background: "#fff",
              cursor: "pointer",
            }}
          >
            Refresh
          </button>
        </div>

        {loading && (
          <div
            style={{
              padding: "40px",
              textAlign: "center",
            }}
          >
            <i
              className="fas fa-spinner fa-spin"
              style={{
                marginRight: "8px",
              }}
            ></i>

            Loading PGs...
          </div>
        )}

        {!loading && error && (
          <div
            style={{
              padding: "25px",
              textAlign: "center",
              border: "1px solid var(--border)",
              borderRadius: "10px",
              background: "#fff",
            }}
          >
            <p
              style={{
                color: "var(--occupied)",
                marginTop: 0,
              }}
            >
              {error}
            </p>

            <button
              type="button"
              onClick={() => loadPGs()}
              style={{
                padding: "8px 14px",
                border: "none",
                borderRadius: "8px",
                background: "var(--accent)",
                color: "#fff",
                cursor: "pointer",
              }}
            >
              Retry
            </button>
          </div>
        )}

        {!loading && !error && pgs.length === 0 && (
          <div
            style={{
              padding: "40px",
              textAlign: "center",
              border: "1px solid var(--border)",
              borderRadius: "10px",
              background: "#fff",
            }}
          >
            <i
              className="fas fa-house-circle-xmark"
              style={{
                fontSize: "30px",
                marginBottom: "10px",
              }}
            ></i>

            <p
              style={{
                margin: 0,
                fontWeight: 600,
              }}
            >
              No PGs found
            </p>

            <p
              style={{
                margin: "6px 0 0",
                color: "var(--muted)",
                fontSize: "12px",
              }}
            >
              Try changing your search filters.
            </p>
          </div>
        )}

        {!loading && !error && pgs.length > 0 && (
          <div
            style={{
              display: "grid",
              gridTemplateColumns:
                "repeat(auto-fill, minmax(320px, 1fr))",
              gap: "16px",
            }}
          >
            {pgs.map((pg) => (
              <PGCard
                key={pg.pg_id}
                pg={pg}
                isSaved={savedPGs.includes(pg.pg_id)}
                isCompared={comparedPGs.includes(pg.pg_id)}
                onToggleSave={toggleSave}
                onToggleCompare={toggleCompare}
                onViewDetails={viewDetails}
              />
            ))}
          </div>
        )}
      </main>
    </div>
  );
}

export default App;
