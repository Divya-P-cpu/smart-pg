import { FI, FL } from "../utils/constants";

function getPgImageUrl(pgId) {
  return `https://picsum.photos/seed/${encodeURIComponent(pgId)}/800/500`;
}

function getGenderLabel(gender) {
  if (!gender) return "Not available";
  return gender;
}

function getAmenityKey(amenity) {
  const value = String(amenity || "").toLowerCase();

  if (value.includes("wi-fi") || value.includes("wifi") || value.includes("internet")) return "wifi";
  if (value.includes("food")) return "food";
  if (value.includes("hot water") || value.includes("geyser")) return "hot_water";
  if (value.includes("washing") || value.includes("laundry")) return "laundry";
  if (value === "ac" || value.includes(" air conditioning")) return "ac";
  if (value.includes("parking")) return "parking";
  if (value.includes("cctv")) return "cctv";
  if (value.includes("lift")) return "lift";
  if (value.includes("power backup")) return "power_backup";
  if (value.includes("housekeeping")) return "housekeeping";
  if (value.includes("study")) return "study_table";
  if (value.includes("locker") || value.includes("personal locker")) return "locker";
  if (value.includes("attached bathroom")) return "attached_bath";

  return "";
}

export default function PGCard({
  pg,
  isSaved,
  isCompared,
  onToggleSave,
  onToggleCompare,
  onViewDetails,
}) {
  const rent = pg?.min_rent;
  const amenities = Array.isArray(pg?.amenities) ? pg.amenities : [];

  return (
    <div
      className="card"
      style={{
        padding: 0,
        overflow: "hidden",
        background: "#fff",
      }}
    >
      <div
        style={{
          position: "relative",
          height: "190px",
          overflow: "hidden",
        }}
      >
        <img
          src={getPgImageUrl(pg?.pg_id)}
          alt={pg?.pg_name || "PG"}
          style={{
            width: "100%",
            height: "100%",
            objectFit: "cover",
            display: "block",
          }}
        />

        {pg?.is_active && (
          <span
            className="badge badge-green"
            style={{
              position: "absolute",
              top: "10px",
              left: "10px",
            }}
          >
            <i
              className="fas fa-circle-check"
              style={{
                fontSize: "9px",
                marginRight: "4px",
              }}
            />
            Available
          </span>
        )}

        <button
          type="button"
          aria-label={isSaved ? "Remove from saved PGs" : "Save PG"}
          onClick={() => onToggleSave(pg.pg_id)}
          style={{
            position: "absolute",
            top: "10px",
            right: "10px",
            width: "36px",
            height: "36px",
            borderRadius: "50%",
            border: "none",
            background: "rgba(255,255,255,0.95)",
            cursor: "pointer",
            display: "flex",
            alignItems: "center",
            justifyContent: "center",
          }}
        >
          <i
            className={`${isSaved ? "fas" : "far"} fa-heart`}
            style={{
              color: isSaved ? "var(--occupied)" : "var(--muted)",
              fontSize: "15px",
            }}
          />
        </button>
      </div>

      <div style={{ padding: "16px" }}>
        <h3
          style={{
            margin: 0,
            fontSize: "16px",
            fontWeight: 800,
            lineHeight: 1.3,
          }}
        >
          {pg?.pg_name || "Unnamed PG"}
        </h3>

        <div
          style={{
            marginTop: "7px",
            fontSize: "12px",
            color: "var(--muted)",
          }}
        >
          <i className="fas fa-location-dot" style={{ marginRight: "5px" }} />
          {pg?.area || "Area not available"}
          {pg?.city && ` · ${pg.city}`}
        </div>

        <div
          style={{
            marginTop: "8px",
            fontSize: "12px",
            color: "var(--muted)",
          }}
        >
          <i className="fas fa-user-group" style={{ marginRight: "5px" }} />
          Gender: {getGenderLabel(pg?.gender_policy)}
        </div>

        <div
          style={{
            display: "flex",
            alignItems: "baseline",
            gap: "5px",
            marginTop: "12px",
          }}
        >
          <span
            style={{
              fontSize: "19px",
              fontWeight: 800,
              color: "var(--fg)",
            }}
          >
            {rent !== null && rent !== undefined && rent !== ""
              ? `₹${Number(rent).toLocaleString("en-IN")}`
              : "Rent not available"}
          </span>

          {rent !== null && rent !== undefined && rent !== "" && (
            <span
              style={{
                fontSize: "11px",
                color: "var(--muted)",
              }}
            >
              /month
            </span>
          )}
        </div>

        <div
          style={{
            display: "flex",
            flexWrap: "wrap",
            gap: "6px",
            marginTop: "12px",
          }}
        >
          {amenities.length > 0 ? (
            amenities.slice(0, 5).map((item, index) => {
              const key = getAmenityKey(item);

              return (
                <span
                  key={`${item}-${index}`}
                  style={{
                    fontSize: "10.5px",
                    color: "var(--muted)",
                    background: "var(--bg)",
                    padding: "5px 7px",
                    borderRadius: "6px",
                  }}
                >
                  <i
                    className={`fas ${FI[key] || "fa-check"}`}
                    style={{
                      fontSize: "9px",
                      marginRight: "3px",
                    }}
                  />
                  {FL[key] || item}
                </span>
              );
            })
          ) : (
            <span
              style={{
                fontSize: "11px",
                color: "var(--muted)",
              }}
            >
              Amenities not available
            </span>
          )}
        </div>
      </div>

      <div
        style={{
          padding: "0 16px 16px",
          display: "flex",
          alignItems: "center",
          gap: "10px",
        }}
      >
        <label
          style={{
            display: "flex",
            alignItems: "center",
            gap: "5px",
            fontSize: "11px",
            color: "var(--muted)",
            cursor: "pointer",
          }}
        >
          <input
            type="checkbox"
            checked={isCompared}
            onChange={() => onToggleCompare(pg.pg_id)}
            style={{
              accentColor: "var(--accent)",
            }}
          />
          Compare
        </label>

        <span style={{ flex: 1 }} />

        <button
          type="button"
          onClick={() => onViewDetails(pg.pg_id)}
          style={{
            background: "var(--accent)",
            color: "#fff",
            border: "none",
            padding: "9px 16px",
            borderRadius: "9px",
            fontFamily: "'Outfit'",
            fontSize: "12px",
            fontWeight: 600,
            cursor: "pointer",
          }}
        >
          VIEW DETAILS
        </button>
      </div>
    </div>
  );
}
