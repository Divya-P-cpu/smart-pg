import React, { useState } from 'react';

export default function OwnerRegister({ onRegisterSuccess, onBackToLogin, onBackToSelection }) {
  const [name, setName] = useState('');
  const [email, setEmail] = useState('');
  const [password, setPassword] = useState('');
  const [pgName, setPgName] = useState('');
  const [error, setError] = useState('');

  const handleSubmit = (e) => {
    e.preventDefault();
    if (!name || !email || !password || !pgName) {
      setError('Please fill in all fields.');
      return;
    }
    
    // Simulate register success
    onRegisterSuccess({
      name: name,
      email: email,
      pgName: pgName,
      role: 'owner'
    });
  };

  return (
    <div className="login-form-container">
      <div className="login-card owner-theme-card">
        <div className="login-header">
          <button className="btn-back-arrow" onClick={onBackToLogin}>
            <i className="fas fa-arrow-left"></i>
          </button>
          <h3>Register as PG Owner</h3>
          <p style={{ fontSize: '12px', color: 'var(--muted)', marginTop: '4px' }}>
            List your properties, configure rooms dynamically, and manage occupant request queues.
          </p>
        </div>

        {error && <div className="auth-error-banner"><i className="fas fa-circle-exclamation"></i> {error}</div>}

        <form onSubmit={handleSubmit} className="auth-form">
          <div className="input-g">
            <label>Owner Name *</label>
            <input
              type="text"
              placeholder="e.g. Sri Sai Manager"
              value={name}
              onChange={(e) => setName(e.target.value)}
              required
            />
          </div>

          <div className="input-g">
            <label>Business Email *</label>
            <input
              type="email"
              placeholder="manager@srisaipg.com"
              value={email}
              onChange={(e) => setEmail(e.target.value)}
              required
            />
          </div>

          <div className="input-g">
            <label>Password *</label>
            <input
              type="password"
              placeholder="••••••••"
              value={password}
              onChange={(e) => setPassword(e.target.value)}
              required
            />
          </div>

          <div className="input-g">
            <label>PG / Co-Living Brand Name *</label>
            <input
              type="text"
              placeholder="e.g. Sri Sai Premium PG"
              value={pgName}
              onChange={(e) => setPgName(e.target.value)}
              required
            />
          </div>

          <button type="submit" className="btn-primary auth-submit-btn owner-submit-btn" style={{ marginTop: '10px' }}>
            REGISTER PG BUSINESS <i className="fas fa-hotel"></i>
          </button>
        </form>

        <div className="auth-footer-help" style={{ marginTop: '20px' }}>
          <p>Already have an account? <span style={{ color: 'var(--accent)', fontWeight: '700', cursor: 'pointer' }} onClick={onBackToLogin}>Log In</span></p>
        </div>
      </div>
    </div>
  );
}
